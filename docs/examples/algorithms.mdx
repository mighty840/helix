## Fibonacci (F(10) = 55)

```asm
.program "fibonacci"
.version 1
.text
.entry main

main:
    IMML R1, 0       ; a = 0
    IMML R2, 1       ; b = 1
    IMML R3, 0       ; i = 0
    IMML R4, 10      ; n = 10

loop:
    CMP  R3, R4
    BGE  done        ; if i >= 10, stop
    ADD  R5, R1, R2  ; t = a + b
    ADD  R1, R2, R0  ; a = b
    ADD  R2, R5, R0  ; b = t
    INC  R3, R3      ; i++
    JMP  loop

done:
    HALT
; R1 = 55
```

**Run:**

```bash
cargo run --bin helix-asm -- examples/fibonacci.hx --no-tmr 2>/dev/null \
  | cargo run --bin helix-vm --
```

## Euclidean GCD (GCD(48, 18) = 6)

```asm
.program "gcd"
.version 1
.text
.entry main

main:
    IMML R1, 48      ; a = 48
    IMML R2, 18      ; b = 18

loop:
    CMP  R1, R2
    BEQ  done
    BLT  swap
    SUB  R1, R1, R2  ; a = a - b  (a > b)
    JMP  loop

swap:
    SUB  R2, R2, R1  ; b = b - a  (b > a)
    JMP  loop

done:
    HALT
; R1 = 6
```

**Run:**

```bash
cargo run --bin helix-asm -- examples/gcd.hx --no-tmr 2>/dev/null \
  | cargo run --bin helix-vm --
```

## Array Sum (sum = 150)

Uses a `.data` section to store the array in memory.

```asm
.program "sum_array"
.version 1
.data
arr: .words 10, 20, 30, 40, 50
.text
.entry main

main:
    IMML R1, 0       ; i = 0
    IMML R2, 5       ; n = 5
    IMML R3, 0       ; sum = 0
    IMML R4, 0       ; base address = 0

loop:
    CMP  R1, R2
    BGE  done
    ADD  R5, R4, R1  ; addr = base + i
    LOAD R6, R5      ; R6 = arr[i]
    ADD  R3, R3, R6  ; sum += arr[i]
    INC  R1, R1      ; i++
    JMP  loop

done:
    HALT
; R3 = 150
```

## Bubble Sort

Sorts an 8-element array in-place using PUSH/POP for the swap:

```asm
.program "bubble_sort"
.version 1
.data
arr: .words 5, 2, 8, 1, 9, 3, 7, 4
.text
.entry main

main:
    IMML R1, 0       ; outer i = 0
    IMML R2, 7       ; outer limit = n-1

outer_check:
    CMP  R1, R2
    BGE  done        ; i >= 7: sorted
    IMML R3, 0       ; inner j = 0
    SUB  R4, R2, R1  ; inner limit = 7 - i

inner_check:
    CMP  R3, R4
    BGE  outer_next
    ADD  R5, R3, R0  ; addr_j = j
    INC  R6, R5      ; addr_j1 = j + 1
    LOAD R5, R5      ; arr[j]
    LOAD R6, R6      ; arr[j+1]
    CMP  R5, R6
    BLT  no_swap
    BEQ  no_swap
    ; swap arr[j] and arr[j+1]
    PUSH R3
    ADD  R7, R3, R0
    STORE R7, R6
    INC  R7, R7
    STORE R7, R5
    POP  R3

no_swap:
    INC  R3, R3
    JMP  inner_check

outer_next:
    INC  R1, R1
    JMP  outer_check

done:
    HALT
```

<Note>
Bubble sort uses 7 registers. Since R0 is hardwired zero, there are only 7 writable registers — this program is at maximum register pressure. PUSH/POP is used to free a register during the swap.
</Note>
